#include <iostream>

using namespace std;

int main() {
    int o1, o2, o3, o4;
    
    if (!(cin >> o1 >> o2 >> o3 >> o4)) {
        return 1;
    }

    bool invalidOctet = (o1 < 1 || o1 > 255) || 
                        (o2 < 1 || o2 > 255) || 
                        (o3 < 1 || o3 > 255) || 
                        (o4 < 1 || o4 > 255);

    if (invalidOctet) {
        cout << "Invalid IP address component\n";
        return 1;
    }

    cout << o1 << "." << o2 << "." << o3 << "." << o4 << endl;

    return 0;
}
