#include <iostream>
#include <cmath>

using namespace std;

int main() {
    double a = 3.0;
    double b = 2.712;

    // Все обчислення в одному рядку
    double y = pow(pow(a, 2) / pow(b, 3), 2.0 / 3.0) * exp((0.807 * (1 - pow(sin((a - b) * M_PI / 4), 2))) / (0.312 * (1 + pow(cos((a + b) * M_PI / 4), 2))));

    cout << "y = " << y << endl;

    return 0;
}
